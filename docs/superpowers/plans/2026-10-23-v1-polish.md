# SkyFallBall v1 Polish Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add 6 polish features (achievements, combo multiplier + Perfect Run, daily missions, FTUE tutorial, rich haptics, share score) to the already-functional SkyFallBall game, feeding one shared coin economy, before App Store submission.

**Architecture:** Three new Angular singleton services (`AchievementService`, `MissionsService`, `TutorialService`) persist state via `@capacitor/preferences`. All three observe the existing `GameEngine.events$` stream — zero new channels. `GameEngine` gains combo-multi state, a `tutorialMode` flag, and three new event kinds (`combo`, `perfectRun`, `speedBrick`). UI additions are inline in existing pages (home gets missions cards + 2 buttons, game gets tutorial overlay + combo HUD) plus one new page (`/achievements`). Haptics extend `AudioService`'s existing `onSoundEvent` dispatcher. Share uses `@capacitor/share`.

**Tech Stack:** Ionic 7, Angular 22 standalone, Capacitor 8, TypeScript 5.4+, Jasmine/Karma, `@capacitor/preferences`, `@capacitor/haptics`, `@capacitor/share`.

- [ ] **Step 2: Create AchievementService**

Create `src/app/core/services/achievement.service.ts`:
```ts
import { Injectable, inject, signal } from '@angular/core';
import { Preferences } from '@capacitor/preferences';
import { WalletService } from './wallet.service';
import { ACHIEVEMENTS } from '../game/achievement-catalog';

const KEY = 'fdb_achievements';

export interface AchievementSignal {
  score?: number;
  coinsThisRun?: number;
  wooowFired?: boolean;
  comboSteps?: number;
  shieldSavedTotal?: number;
  dailyStreak?: number;
  runEnded?: boolean;
}

@Injectable({ providedIn: 'root' })
export class AchievementService {
  private wallet = inject(WalletService);
  readonly unlocked = signal<Set<string>>(new Set());
  readonly progress = signal<Record<string, number>>({});

  async load(): Promise<void> {
    try {
      const { value } = await Preferences.get({ key: KEY });
      if (!value) return;
      const d = JSON.parse(value);
      if (Array.isArray(d.unlocked)) this.unlocked.set(new Set(d.unlocked));
      if (d.progress && typeof d.progress === 'object') this.progress.set(d.progress);
    } catch (e) { console.warn('AchievementService.load failed', e); }
  }

  async save(): Promise<void> {
    await Preferences.set({
      key: KEY,
      value: JSON.stringify({ unlocked: [...this.unlocked()], progress: this.progress() }),
    });
  }

  recordEvent(sig: AchievementSignal): string[] {
    const newly: string[] = [];
    const unlocked = new Set(this.unlocked());
    const progress = { ...this.progress() };

    for (const def of ACHIEVEMENTS) {
      if (unlocked.has(def.id)) continue;
      if (this.triggerMatches(def.trigger, sig, progress)) {
        unlocked.add(def.id);
        newly.push(def.id);
        this.wallet.addCoins(def.reward);
      }
    }

    if (newly.length) {
      this.unlocked.set(unlocked);
      this.progress.set(progress);
      this.save();
    }
    return newly;
  }

  private triggerMatches(
    t: ACHIEVEMENTS[number]['trigger'],
    s: AchievementSignal,
    progress: Record<string, number>,
  ): boolean {
    switch (t.kind) {
      case 'first_run_ended':   return !!s.runEnded;
      case 'score_run':         return (s.score ?? 0) >= t.min;
      case 'coins_run':         return (s.coinsThisRun ?? 0) >= t.min;
      case 'coins_lifetime':    return this.wallet.coins() >= t.min;
      case 'shield_saved_lifetime': return (s.shieldSavedTotal ?? 0) >= t.min;
      case 'first_wooow':       return !!s.wooowFired;
      case 'combo_steps':       return (s.comboSteps ?? 0) >= t.min;
      case 'all_skins_unlocked': return this.wallet.unlockedSkins().size >= t.totalSkins;
      case 'daily_streak':      return (s.dailyStreak ?? 0) >= t.min;
    }
  }
}
```

  { type: 'coins_single',  targets: [30, 50, 80],    reward: 40, nameKey: 'mission.coins_single',  icon: '💰' },
  { type: 'runs_day',      targets: [3, 5, 10],      reward: 40, nameKey: 'mission.runs_day',      icon: '🎮' },
  { type: 'combos_day',    targets: [3, 5, 10],      reward: 40, nameKey: 'mission.combos_day',    icon: '⚡' },
  { type: 'powerups_day',  targets: [3, 5, 10],      reward: 40, nameKey: 'mission.powerups_day',  icon: '🛡️' },
  { type: 'survive_single',targets: [60, 90, 120],   reward: 40, nameKey: 'mission.survive_single',icon: '⏱️' },

export interface Mission {
  id: string;
  type: MissionType;
  target: number;
  progress: number;
  claimed: boolean;
}

export function todayKey(d: Date = new Date()): string {
  const y = d.getFullYear(), m = String(d.getMonth() + 1).padStart(2, '0'), day = String(d.getDate()).padStart(2, '0');
  return `${y}-${m}-${day}`;
}

export function rollDailyMissions(templates: MissionTemplate[] = MISSION_TEMPLATES, rand: () => number = Math.random): Mission[] {
  const pool = [...templates];
  const picked: MissionTemplate[] = [];
  for (let i = 0; i < 3 && pool.length; i++) {
    const idx = Math.floor(rand() * pool.length);
    picked.push(pool.splice(idx, 1)[0]);
  }
  return picked.map((t, i) => ({
    id: `${todayKey()}-${t.type}-${i}`,
    type: t.type,
    target: t.targets[Math.floor(rand() * t.targets.length)],
    progress: 0,
    claimed: false,
  }));
}

```ts
import { Injectable, inject, signal } from '@angular/core';
import { Preferences } from '@capacitor/preferences';
import { WalletService } from './wallet.service';
import {
  Mission, MissionType, MISSION_TEMPLATES, rollDailyMissions, todayKey,
} from '../game/missions-catalog';

const KEY = 'fdb_missions';

export type MissionEventKind = 'run_end' | 'coin_run' | 'combo' | 'powerup' | 'survive_run';

interface State { dayKey: string; missions: Mission[]; streak: number; }

@Injectable({ providedIn: 'root' })
export class MissionsService {
  private wallet = inject(WalletService);
  readonly missions = signal<Mission[]>([]);
  readonly streak = signal(0);
  private dayKey = '';

  async load(): Promise<void> {
    try {
      const { value } = await Preferences.get({ key: KEY });
      if (value) {
        const d = JSON.parse(value) as State;
        this.dayKey = d.dayKey;
        this.missions.set(d.missions);
        this.streak.set(d.streak);
      }
    } catch (e) { console.warn('MissionsService.load failed', e); }
    this.refreshIfNewDay();
  }

  private refreshIfNewDay(): void {
    const today = todayKey();
    if (this.dayKey === today) return;
    const allClaimed = this.missions().length === 3 && this.missions().every(m => m.claimed);
    this.streak.set(allClaimed ? this.streak() + 1 : 0);
    this.dayKey = today;
    this.missions.set(rollDailyMissions());
    this.save();
  }

  recordEvent(kind: MissionEventKind, value: number = 1): void {
    this.refreshIfNewDay();
    const updated = this.missions().map(m => {
      if (m.claimed) return m;
      const match =
        (kind === 'run_end' && m.type === 'score_single' && value >= m.target) ? m.target :
        (kind === 'coin_run' && m.type === 'coins_single') ? Math.max(m.progress, value) :
        (kind === 'survive_run' && m.type === 'survive_single') ? Math.max(m.progress, value) :
        (kind === 'run_end' && m.type === 'runs_day') ? m.progress + 1 :
        (kind === 'combo' && m.type === 'combos_day') ? m.progress + 1 :
        (kind === 'powerup' && m.type === 'powerups_day') ? m.progress + 1 :
        null;
      if (match === null) return m;
      return { ...m, progress: Math.min(match, m.target) };
    });
    this.missions.set(updated);
    this.save();
  }

  claim(id: string): boolean {
    const m = this.missions().find(x => x.id === id);
    if (!m || m.claimed || m.progress < m.target) return false;
    this.wallet.addCoins(m.reward);
    this.missions.set(this.missions().map(x => x.id === id ? { ...x, claimed: true } : x));
    this.save();
    return true;
  }

  async save(): Promise<void> {
    const state: State = { dayKey: this.dayKey, missions: this.missions(), streak: this.streak() };
    await Preferences.set({ key: KEY, value: JSON.stringify(state) });
  }
}

import { Injectable, signal } from '@angular/core';
import { Preferences } from '@capacitor/preferences';

const KEY = 'fdb_tutorial_done';

@Injectable({ providedIn: 'root' })
export class TutorialService {
  readonly step = signal(0);     // 0 = inactive, 1..4 = steps, 5 = done this run
  readonly active = signal(false);
  private done = false;

  async load(): Promise<void> {
    try {
      const { value } = await Preferences.get({ key: KEY });
      this.done = value === 'true';
    } catch { this.done = false; }
  }

  async markDone(): Promise<void> {
    this.done = true;
    this.active.set(false);
    this.step.set(0);
    await Preferences.set({ key: KEY, value: 'true' });
  }

  async resetForReplay(): Promise<void> {
    this.done = false;
    await Preferences.remove({ key: KEY });
  }

  shouldLaunch(): boolean { return !this.done; }

  start(): void {
    this.active.set(true);
    this.step.set(1);
  }

  advanceStep(): void {
    const next = this.step() + 1;
    if (next > 4) { this.markDone(); return; }
    this.step.set(next);
  }

  skip(): Promise<void> { return this.markDone(); }
}
```
